Sinkronisasi Google Sheets (Pantau Bersama Istri)

Hubungkan aplikasi ini ke Google Sheets agar istri di Kasembon bisa memantau saldo Pos Dapur, WiFi, Modal Putar, dan Cicilan Bank secara real-time.

Kode Google Apps Script (Code.gs): 
function getOrCreateSheet_(ss, name, headers) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
  }
  if (headers && headers.length > 0 && sheet.getLastRow() === 0) {
    sheet.appendRow(headers);
  }
  return sheet;
}

function jsonResponse_(payload) {
  return ContentService
    .createTextOutput(JSON.stringify(payload))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var logsSheet = getOrCreateSheet_(ss, "Logs", [
      "Waktu",
      "Jenis",
      "Kategori/Pos",
      "Keterangan",
      "Nominal (Rp)"
    ]);
    var stateSheet = getOrCreateSheet_(ss, "state", [
      "Kunci",
      "Diperbarui Pada",
      "Payload JSON"
    ]);

    var raw = e && e.postData && e.postData.contents ? e.postData.contents : "{}";
    var data = JSON.parse(raw);
    var action = data.action || (data.state ? "save_state" : "append_log");

    var logged = false;
    var stateSaved = false;

    if (data.entry || action === "append_log") {
      var item = data.entry || data;
      logsSheet.appendRow([
        new Date(),
        item.type || "Log",
        item.category || "-",
        item.text || "-",
        Number(item.amount || 0)
      ]);
      logged = true;
    }

    if (data.state || action === "save_state") {
      var snapshotObj = data.state || data.payload || {};
      var snapshotJson = JSON.stringify(snapshotObj);
      var now = new Date();

      var lastRow = stateSheet.getLastRow();
      var targetRow = -1;

      if (lastRow >= 2) {
        var keys = stateSheet.getRange(2, 1, lastRow - 1, 1).getValues();
        for (var i = 0; i < keys.length; i++) {
          if (String(keys[i][0]) === "current_state") {
            targetRow = i + 2;
            break;
          }
        }
      }

      if (targetRow > 0) {
        stateSheet.getRange(targetRow, 1, 1, 3).setValues([
          ["current_state", now, snapshotJson]
        ]);
      } else {
        stateSheet.appendRow(["current_state", now, snapshotJson]);
      }

      stateSaved = true;
    }

    return jsonResponse_({
      status: "ok",
      action: action,
      logged: logged,
      stateSaved: stateSaved
    });
  } catch (err) {
    return jsonResponse_({
      status: "error",
      message: err && err.message ? err.message : String(err)
    });
  }
}

function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var stateSheet = ss.getSheetByName("state");
    var logsSheet = ss.getSheetByName("Logs");

    var stateObj = null;
    var updatedAt = null;

    if (stateSheet && stateSheet.getLastRow() >= 2) {
      var rows = stateSheet.getRange(2, 1, stateSheet.getLastRow() - 1, 3).getValues();
      for (var i = rows.length - 1; i >= 0; i--) {
        if (String(rows[i][0]) === "current_state" && rows[i][2]) {
          stateObj = JSON.parse(String(rows[i][2]));
          updatedAt = rows[i][1];
          break;
        }
      }
    }

    var recentLogs = [];
    if (logsSheet && logsSheet.getLastRow() >= 2) {
      var totalRows = logsSheet.getLastRow() - 1;
      var limit = Math.min(30, totalRows);
      var startRow = logsSheet.getLastRow() - limit + 1;
      var logValues = logsSheet.getRange(startRow, 1, limit, 5).getValues();

      for (var j = logValues.length - 1; j >= 0; j--) {
        recentLogs.push({
          time: logValues[j][0],
          type: logValues[j][1],
          category: logValues[j][2],
          text: logValues[j][3],
          amount: Number(logValues[j][4] || 0)
        });
      }
    }

    return jsonResponse_({
      status: "ok",
      message: "Pocket Command Web App Aktif",
      updatedAt: updatedAt,
      state: stateObj,
      logs: recentLogs
    });
  } catch (err) {
    return jsonResponse_({
      status: "error",
      message: err && err.message ? err.message : String(err)
    });
  }
}
